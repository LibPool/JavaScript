# react-storeprops

> 标签: JavaScript

## 简介

```jsx interface Props {     value: string;     onChange: (x: string) => void; } class MyComp extends React.PureComponent<Props, {}> {     render() {         return <input value={this.props.value} onChange={this.props.onChange} />     } } ```
